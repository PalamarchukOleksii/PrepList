import 'package:flutter/material.dart';

import 'package:preplist/features/foods/repositories/food_repository.dart';
import 'package:preplist/data/services/swiss_food_import_service.dart';

class FoodBootstrapper extends StatefulWidget {
  final Widget child;

  const FoodBootstrapper({super.key, required this.child});

  @override
  State<FoodBootstrapper> createState() => _FoodBootstrapperState();
}

class _FoodBootstrapperState extends State<FoodBootstrapper> {
  bool _loading = true;
  bool _failed = false;

  int _completed = 0;
  int _total = 0;

  String? _error;

  @override
  void initState() {
    super.initState();
    _initialize();
  }

  Future<void> _initialize() async {
    setState(() {
      _loading = true;
      _failed = false;
      _error = null;
      _completed = 0;
      _total = 0;
    });

    try {
      final database = FoodRepository.instance;

      if (await database.isImported()) {
        if (!mounted) return;

        setState(() => _loading = false);
        return;
      }

      final importer = SwissFoodImportService();

      await importer.importDatabase(
        onProgress: (completed, total) {
          if (!mounted) return;

          setState(() {
            _completed = completed;
            _total = total;
          });
        },
      );

      if (!mounted) return;

      setState(() => _loading = false);
    } catch (error) {
      if (!mounted) return;

      setState(() {
        _loading = false;
        _failed = true;
        _error = error.toString();
      });
    }
  }

  @override
  Widget build(BuildContext context) {
    if (!_loading && !_failed) {
      return widget.child;
    }

    final progress = _total == 0 ? null : _completed / _total;

    return Scaffold(
      body: SafeArea(
        child: Center(
          child: Padding(
            padding: const EdgeInsets.all(24),
            child: ConstrainedBox(
              constraints: const BoxConstraints(maxWidth: 420),
              child: Column(
                mainAxisAlignment: MainAxisAlignment.center,
                children: [
                  DecoratedBox(
                    decoration: BoxDecoration(
                      borderRadius: BorderRadius.circular(24),
                      boxShadow: [
                        BoxShadow(
                          color: Colors.black.withValues(alpha: 0.15),
                          blurRadius: 12,
                          offset: const Offset(0, 4),
                        ),
                      ],
                    ),
                    child: ClipRRect(
                      borderRadius: BorderRadius.circular(24),
                      child: Image.asset(
                        'assets/icon/app_icon.png',
                        width: 96,
                        height: 96,
                        fit: BoxFit.cover,
                      ),
                    ),
                  ),
                  const SizedBox(height: 24),
                  Text(
                    _failed
                        ? 'Could not download food database'
                        : 'Preparing your food database',
                    style: Theme.of(context).textTheme.headlineSmall,
                    textAlign: TextAlign.center,
                  ),
                  const SizedBox(height: 12),
                  Text(
                    _failed
                        ? 'An internet connection is required for the '
                              'initial download.'
                        : 'Downloading the Swiss Food Composition Database. '
                              'This is only required once.',
                    textAlign: TextAlign.center,
                  ),
                  const SizedBox(height: 24),
                  if (_loading) ...[
                    LinearProgressIndicator(value: progress),
                    const SizedBox(height: 12),
                    Text(
                      _total == 0
                          ? 'Connecting to Swiss database...'
                          : '$_completed / $_total foods',
                    ),
                  ],
                  if (_failed) ...[
                    Text(
                      _error ?? 'Unknown error',
                      textAlign: TextAlign.center,
                      style: TextStyle(
                        color: Theme.of(context).colorScheme.error,
                      ),
                    ),
                    const SizedBox(height: 20),
                    FilledButton(
                      onPressed: _initialize,
                      child: const Text('Try again'),
                    ),
                  ],
                ],
              ),
            ),
          ),
        ),
      ),
    );
  }
}

import 'package:path/path.dart' as path;
import 'package:sqflite/sqflite.dart';

class AppDatabase {
  AppDatabase._();

  static final AppDatabase instance = AppDatabase._();

  Database? _database;

  Future<Database> get database async {
    if (_database != null) return _database!;

    final databasePath = await getDatabasesPath();

    _database = await openDatabase(
      path.join(databasePath, 'preplist.db'),
      version: 8,
      onConfigure: (db) async {
        await db.execute('PRAGMA foreign_keys = ON');
      },
      onCreate: (db, version) async {
        await _createMealsTable(db);
        await _createRecipeTables(db);
        await _createShoppingTables(db);
        await _createInventoryUsageTable(db);
        await _createFoodTables(db);
      },
      onUpgrade: (db, oldVersion, newVersion) async {
        if (oldVersion < 2) {
          await _createRecipeTables(db);
        }

        if (oldVersion < 3) {
          await db.execute(
            "ALTER TABLE meals ADD COLUMN item_type TEXT NOT NULL DEFAULT 'ingredient'",
          );
          await db.execute('ALTER TABLE meals ADD COLUMN recipe_id INTEGER');
          await db.execute(
            "ALTER TABLE meals ADD COLUMN quantity TEXT NOT NULL DEFAULT ''",
          );
          await db.execute(
            "ALTER TABLE meals ADD COLUMN unit TEXT NOT NULL DEFAULT ''",
          );
        }

        if (oldVersion < 4) {
          await _createShoppingTables(db);
        }

        if (oldVersion >= 4 && oldVersion < 5) {
          await db.execute(
            'ALTER TABLE grocery_items '
            'ADD COLUMN inventory_added INTEGER NOT NULL DEFAULT 0',
          );
        }

        if (oldVersion < 6) {
          await db.execute(
            'ALTER TABLE recipe_ingredients ADD COLUMN food_id INTEGER',
          );
          await db.execute('ALTER TABLE meals ADD COLUMN food_id INTEGER');
        }

        if (oldVersion < 7) {
          await _createInventoryUsageTable(db);
        }

        if (oldVersion < 8) {
          await _createFoodTables(db);
        }
      },
    );

    return _database!;
  }

  Future<void> _createMealsTable(DatabaseExecutor db) async {
    await db.execute('''
      CREATE TABLE meals (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        date TEXT NOT NULL,
        category TEXT NOT NULL,
        name TEXT NOT NULL,
        item_type TEXT NOT NULL DEFAULT 'ingredient',
        recipe_id INTEGER,
        food_id INTEGER,
        quantity TEXT NOT NULL DEFAULT '',
        unit TEXT NOT NULL DEFAULT ''
      )
    ''');

    await db.execute('''
      CREATE INDEX meals_date_index ON meals(date)
    ''');
  }

  Future<void> _createRecipeTables(DatabaseExecutor db) async {
    await db.execute('''
      CREATE TABLE recipes (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        name TEXT NOT NULL,
        description TEXT NOT NULL DEFAULT '',
        instructions TEXT NOT NULL DEFAULT '',
        prep_minutes INTEGER NOT NULL DEFAULT 0,
        cook_minutes INTEGER NOT NULL DEFAULT 0,
        servings INTEGER NOT NULL DEFAULT 1,
        calories REAL NOT NULL DEFAULT 0,
        protein REAL NOT NULL DEFAULT 0,
        carbs REAL NOT NULL DEFAULT 0,
        fat REAL NOT NULL DEFAULT 0
      )
    ''');

    await db.execute('''
      CREATE TABLE recipe_ingredients (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        recipe_id INTEGER NOT NULL,
        food_id INTEGER,
        name TEXT NOT NULL,
        amount TEXT NOT NULL DEFAULT '',
        unit TEXT NOT NULL DEFAULT '',
        FOREIGN KEY (recipe_id)
          REFERENCES recipes(id)
          ON DELETE CASCADE
      )
    ''');

    await db.execute('''
      CREATE INDEX recipe_ingredients_recipe_index
      ON recipe_ingredients(recipe_id)
    ''');
  }

  Future<void> _createShoppingTables(DatabaseExecutor db) async {
    await db.execute('''
      CREATE TABLE grocery_items (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        item_key TEXT NOT NULL UNIQUE,
        name TEXT NOT NULL,
        amount TEXT NOT NULL DEFAULT '',
        unit TEXT NOT NULL DEFAULT '',
        is_bought INTEGER NOT NULL DEFAULT 0,
        inventory_added INTEGER NOT NULL DEFAULT 0
      )
    ''');

    await db.execute('''
      CREATE TABLE inventory_items (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        item_key TEXT NOT NULL UNIQUE,
        name TEXT NOT NULL,
        amount TEXT NOT NULL DEFAULT '',
        unit TEXT NOT NULL DEFAULT ''
      )
    ''');
  }

  Future<void> _createInventoryUsageTable(DatabaseExecutor db) async {
    await db.execute('''
      CREATE TABLE inventory_usage (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        item_key TEXT NOT NULL,
        name TEXT NOT NULL,
        amount TEXT NOT NULL,
        unit TEXT NOT NULL DEFAULT '',
        used_at TEXT NOT NULL,
        is_undone INTEGER NOT NULL DEFAULT 0
      )
    ''');

    await db.execute('''
      CREATE INDEX inventory_usage_date_index
      ON inventory_usage(used_at)
    ''');
  }

  Future<void> _createFoodTables(DatabaseExecutor db) async {
    await db.execute('''
      CREATE TABLE foods (
        external_id INTEGER PRIMARY KEY,
        name TEXT NOT NULL,
        synonyms TEXT NOT NULL DEFAULT '',
        calories REAL NOT NULL DEFAULT 0,
        protein REAL NOT NULL DEFAULT 0,
        carbs REAL NOT NULL DEFAULT 0,
        fat REAL NOT NULL DEFAULT 0
      )
    ''');

    await db.execute('''
      CREATE INDEX foods_name_index ON foods(name)
    ''');

    await db.execute('''
      CREATE TABLE metadata (
        key TEXT PRIMARY KEY,
        value TEXT NOT NULL
      )
    ''');
  }
}

import 'dart:async';
import 'dart:convert';

import 'package:http/http.dart' as http;

class SwissFoodApi {
  static const String _baseUrl =
      'https://api.webapp.prod.blv.foodcase-services.com/BLV_WebApp_WS';

  static const String _apiPath = '/webresources/BLV-api';

  final http.Client _client;

  SwissFoodApi({http.Client? client}) : _client = client ?? http.Client();

  Future<String> fetchVersion() async {
    final json = await _getJson('$_apiPath/versiondb');

    if (json is Map) {
      return (json['versiontext'] ?? json['idversion'] ?? 'unknown').toString();
    }

    return 'unknown';
  }

  Future<List<Map<String, dynamic>>> fetchFoodPage({
    required int offset,
    int limit = 100,
  }) async {
    final json = await _getJson(
      '$_apiPath/foods',
      query: {'lang': 'en', 'limit': '$limit', 'offset': '$offset'},
    );

    final items = _extractList(json);

    return items
        .whereType<Map>()
        .map((item) => Map<String, dynamic>.from(item))
        .toList();
  }

  Future<Map<String, dynamic>> fetchFoodDetails(int id) async {
    final json = await _getJson('$_apiPath/food/$id', query: {'lang': 'en'});

    if (json is! Map) {
      throw FormatException('Invalid food response for ID $id');
    }

    return Map<String, dynamic>.from(json);
  }

  Future<dynamic> _getJson(String path, {Map<String, String>? query}) async {
    final uri = Uri.parse('$_baseUrl$path').replace(queryParameters: query);

    Object? lastError;

    for (var attempt = 0; attempt < 4; attempt++) {
      try {
        final response = await _client
            .get(uri, headers: const {'Accept': 'application/json'})
            .timeout(const Duration(seconds: 30));

        if (response.statusCode >= 200 && response.statusCode < 300) {
          return jsonDecode(utf8.decode(response.bodyBytes));
        }

        if (response.statusCode == 429 || response.statusCode >= 500) {
          lastError = 'HTTP ${response.statusCode}: ${response.body}';
          await Future.delayed(Duration(seconds: 2 * (attempt + 1)));
          continue;
        }

        throw Exception(
          'Swiss database returned HTTP ${response.statusCode}: '
          '${response.body}',
        );
      } on TimeoutException catch (error) {
        lastError = error;
      } catch (error) {
        if (attempt == 3) rethrow;
        lastError = error;
        await Future.delayed(Duration(seconds: 2 * (attempt + 1)));
      }
    }

    throw Exception('Swiss database request failed: $lastError');
  }

  List<dynamic> _extractList(dynamic json) {
    if (json is List) return json;

    if (json is Map) {
      for (final key in ['foods', 'items', 'data', 'results', 'content']) {
        final value = json[key];
        if (value is List) return value;
      }
    }

    throw const FormatException(
      'Could not find a food list in the Swiss database response.',
    );
  }

  void close() {
    _client.close();
  }
}

class SwissFoodImportService {

  final FoodRepository repository;

  SwissFoodImportService({SwissFoodApi? api, FoodRepository? repository})
    : api = api ?? SwissFoodApi(),

  Future<void> importDatabase({required ImportProgress onProgress}) async {
    final version = await api.fetchVersion();

    const pageSize = 100;
    var offset = 0;

    final foodList = <Map<String, dynamic>>[];

    while (true) {
      final page = await api.fetchFoodPage(offset: offset, limit: pageSize);

      foodList.addAll(page);

      if (page.length < pageSize) break;

      offset += pageSize;
    }

    if (foodList.isEmpty) {
      throw Exception('The Swiss database returned no foods.');
    }

    final ids = foodList.map(_extractId).whereType<int>().toList();

    if (ids.isEmpty) {
      throw Exception('No food IDs were found in the API response.');
    }

    final records = <Map<String, dynamic>>[];
    var completed = 0;

    onProgress(0, ids.length);

    // A limited number of concurrent requests.
    const workerCount = 5;
    var nextIndex = 0;

    Future<void> worker() async {
      while (true) {
        final index = nextIndex++;

        if (index >= ids.length) return;

        final id = ids[index];
        final details = await api.fetchFoodDetails(id);

        final record = _parseFood(details, id);

        if (record != null) {
          records.add(record);
        }

        completed++;
        onProgress(completed, ids.length);
      }
    }

    await Future.wait(List.generate(workerCount, (_) => worker()));

    if (records.isEmpty) {
      throw Exception('No valid food records were downloaded.');
    }

    await repository.replaceAllFoods(foods: records, version: version);
  }

  int? _extractId(Map<String, dynamic> food) {
    final value = food['id'] ?? food['DBID'] ?? food['dbid'] ?? food['foodid'];

    if (value is int) return value;
    return int.tryParse(value?.toString() ?? '');
  }

  Map<String, dynamic>? _parseFood(Map<String, dynamic> food, int fallbackId) {
    final id = _extractId(food) ?? fallbackId;

    final name = _extractName(food);

    if (name.isEmpty) return null;

    final synonyms = _extractSynonyms(food);
    final values = food['values'];

    double calories = 0;
    double protein = 0;
    double carbs = 0;
    double fat = 0;

    if (values is List) {
      for (final item in values) {
        if (item is! Map) continue;

        final value = _toDouble(item['value']);
        if (value == null) continue;

        final component = item['component'];
        final componentMap = component is Map ? component : <String, dynamic>{};

        final componentName =
            '${componentMap['name'] ?? item['componentname'] ?? ''}'
                .toLowerCase();

        final componentCode = '${componentMap['code'] ?? item['code'] ?? ''}'
            .toLowerCase();

        final unit = '${item['unit'] ?? componentMap['unit'] ?? ''}'
            .toLowerCase();

        final searchable = '$componentName $componentCode';

        if (_matches(searchable, ['protein', 'proteins', 'prot'])) {
          protein = value;
        } else if (_matches(searchable, [
          'carbohydrate',
          'carbohydrates',
          'available carbohydrate',
          'carb',
          'cho',
        ])) {
          carbs = value;
        } else if (_matches(searchable, [
          'fat',
          'total fat',
          'lipid',
          'lipids',
        ])) {
          fat = value;
        } else if (_matches(searchable, ['energy', 'ener'])) {
          calories = unit.contains('kj') ? value / 4.184 : value;
        }
      }
    }

    return {
      'external_id': id,
      'name': name,
      'synonyms': synonyms.join('|'),
      'calories': calories,
      'protein': protein,
      'carbs': carbs,
      'fat': fat,
    };
  }

  String _extractName(Map<String, dynamic> food) {
    final direct = food['name'];

    if (direct is String && direct.trim().isNotEmpty) {
      return direct.trim();
    }

    final names = food['names'];

    if (names is List) {
      for (final item in names) {
        if (item is String && item.trim().isNotEmpty) {
          return item.trim();
        }

        if (item is Map) {
          final value = item['name'] ?? item['value'];
          if (value != null && value.toString().trim().isNotEmpty) {
            return value.toString().trim();
          }
        }
      }
    }

    if (names is Map) {
      final english = names['en'];
      if (english != null) return english.toString();
    }

    return '';
  }

  List<String> _extractSynonyms(Map<String, dynamic> food) {
    final value = food['synonyms'] ?? food['synonymes'];

    if (value is List) {
      return value
          .map((item) {
            if (item is Map) {
              return (item['name'] ?? item['value'] ?? '').toString();
            }
            return item.toString();
          })
          .where((item) => item.trim().isNotEmpty)
          .toList();
    }

    return [];
  }

  bool _matches(String text, List<String> terms) {
    return terms.any((term) => text.contains(term));
  }

  double? _toDouble(dynamic value) {
    if (value is num) return value.toDouble();
    if (value == null) return null;

    return double.tryParse(value.toString().replaceAll(',', '.'));
  }
}

      calories: _toDouble(map['calories']),
      protein: _toDouble(map['protein']),

      fat: _toDouble(map['fat']),

    return double.tryParse(value?.toString() ?? '') ?? 0;
  }

class _GroceryDraft {
  final String name;
  final String unit;
  double? numericAmount;
  String amount;

  _GroceryDraft({
    required this.name,
    required this.unit,
    required this.amount,
    this.numericAmount,
  });

class GroceryRepository {
  final AppDatabase _database = AppDatabase.instance;
  final MealRepository _mealRepository = MealRepository();
  final RecipeRepository _recipeRepository = RecipeRepository();

  String _normalize(String value) => value.trim().toLowerCase();

  String _makeKey(String name, String unit) {
    return '${_normalize(name)}|${_normalize(unit)}';
  }

  double? _parseAmount(String value) {
    return double.tryParse(value.trim().replaceAll(',', '.'));
  }

  String _formatAmount(double value) {
    if (value == value.roundToDouble()) {
      return value.toInt().toString();
    }

    return value.toStringAsFixed(2).replaceFirst(RegExp(r'\.?0+$'), '');
  }

  Future<List<GroceryItem>> getAll() async {
    final db = await _database.database;

    final rows = await db.query(
      'grocery_items',
      orderBy: 'is_bought ASC, name COLLATE NOCASE ASC',
    );

    return rows.map(GroceryItem.fromMap).toList();
  }

  Future<void> syncFromFuturePlan() async {
    final meals = await _mealRepository.getAll();
    final recipes = <int, Recipe>{};

    final today = _dateKey(DateTime.now());
    final futureMeals = meals.where((meal) => meal.date.compareTo(today) >= 0);

    final grouped = <String, _GroceryDraft>{};

    void addIngredient(String name, String amount, String unit) {
      final cleanName = name.trim();
      if (cleanName.isEmpty) return;

      final cleanUnit = unit.trim();
      final key = _makeKey(cleanName, cleanUnit);
      final parsedAmount = _parseAmount(amount);
      final existing = grouped[key];

      if (existing == null) {
        grouped[key] = _GroceryDraft(
          name: cleanName,
          unit: cleanUnit,
          amount: amount.trim(),
          numericAmount: parsedAmount,
        );
        return;
      }

      if (existing.numericAmount != null && parsedAmount != null) {
        existing.numericAmount = existing.numericAmount! + parsedAmount;
        existing.amount = _formatAmount(existing.numericAmount!);
      } else if (existing.amount.isEmpty) {
        existing.amount = amount.trim();
      }
    }

    for (final meal in futureMeals) {
      if (meal.isRecipe && meal.recipeId != null) {
        final recipeId = meal.recipeId!;

        final recipe =
            recipes[recipeId] ?? await _recipeRepository.getById(recipeId);

        if (recipe == null) continue;

        recipes[recipeId] = recipe;

        final multiplier = _parseAmount(meal.quantity) ?? 1;

        for (final ingredient in recipe.ingredients) {
          final baseAmount = _parseAmount(ingredient.amount);

          final amount = baseAmount == null
              ? ingredient.amount
              : _formatAmount(baseAmount * multiplier);

          addIngredient(ingredient.name, amount, ingredient.unit);
        }
      } else {
        addIngredient(meal.name, meal.quantity, meal.unit);
      }
    }

    final db = await _database.database;

    await db.transaction((txn) async {
      final existingRows = await txn.query('grocery_items');

      final existingByKey = <String, Map<String, dynamic>>{
        for (final row in existingRows) row['item_key'] as String: row,
      };

      // Get items already available in inventory.
      final inventoryRows = await txn.query('inventory_items');

      final inventoryKeys = inventoryRows
          .map((row) => row['item_key'] as String)
          .toSet();

      // Remove old, unbought grocery items.
      await txn.delete('grocery_items', where: 'is_bought = 0');

      for (final entry in grouped.entries) {
        final key = entry.key;
        final draft = entry.value;
        final previous = existingByKey[key];

        // Don't recreate items already in inventory.
        if (inventoryKeys.contains(key)) continue;

        final amount = draft.numericAmount == null
            ? draft.amount
            : _formatAmount(draft.numericAmount!);

        if (previous != null && (previous['is_bought'] as int? ?? 0) == 1) {
          await txn.update(
            'grocery_items',
            {'name': draft.name, 'amount': amount, 'unit': draft.unit},
            where: 'item_key = ?',
            whereArgs: [key],
          );
        } else {
          await txn.insert('grocery_items', {
            'item_key': key,
            'name': draft.name,
            'amount': amount,
            'unit': draft.unit,
            'is_bought': 0,
            'inventory_added': 0,
          });
        }
      }
    });
  }

  Future<void> toggleBought(int groceryId) async {
    final db = await _database.database;

    await db.rawUpdate(
      '''
      UPDATE grocery_items
      SET is_bought = CASE WHEN is_bought = 1 THEN 0 ELSE 1 END
      WHERE id = ?
      ''',
      [groceryId],
    );
  }

  Future<void> addBoughtToInventory(int groceryId) async {
    final db = await _database.database;

    await db.transaction((txn) async {
      final rows = await txn.query(
        'grocery_items',
        where: 'id = ?',
        whereArgs: [groceryId],
        limit: 1,
      );

      if (rows.isEmpty) return;

      final grocery = rows.first;

      final isBought = (grocery['is_bought'] as int? ?? 0) == 1;
      final alreadyAdded = (grocery['inventory_added'] as int? ?? 0) == 1;

      if (!isBought || alreadyAdded) return;

      final itemKey = grocery['item_key'] as String;
      final amount = grocery['amount'] as String? ?? '';
      final unit = grocery['unit'] as String? ?? '';

      final inventoryRows = await txn.query(
        'inventory_items',
        where: 'item_key = ?',
        whereArgs: [itemKey],
        limit: 1,
      );

      if (inventoryRows.isEmpty) {
        await txn.insert('inventory_items', {
          'item_key': itemKey,
          'name': grocery['name'],
          'amount': amount,
          'unit': unit,
        });
      } else {
        final oldAmount = inventoryRows.first['amount'] as String? ?? '';
        final oldNumber = _parseAmount(oldAmount);
        final newNumber = _parseAmount(amount);

        final combinedAmount = oldNumber != null && newNumber != null
            ? _formatAmount(oldNumber + newNumber)
            : amount.isNotEmpty
            ? amount
            : oldAmount;

        await txn.update(
          'inventory_items',
          {'amount': combinedAmount, 'unit': unit},
          where: 'item_key = ?',
          whereArgs: [itemKey],
        );
      }

      // Remove the item from groceries after inventory transfer.
      await txn.delete(
        'grocery_items',
        where: 'id = ?',
        whereArgs: [groceryId],
      );
    });
  }

  String _dateKey(DateTime date) {
    final year = date.year.toString().padLeft(4, '0');
    final month = date.month.toString().padLeft(2, '0');
    final day = date.day.toString().padLeft(2, '0');

    return '$year-$month-$day';
  }
}

    );
  }

class InventoryUsage {
  final int id;
  final String itemKey;
  final String name;
  final String amount;
  final String unit;
  final DateTime usedAt;
  final bool isUndone;

  const InventoryUsage({
    required this.id,
    required this.itemKey,
    required this.name,
    required this.amount,
    required this.unit,
    required this.usedAt,
    required this.isUndone,
  });

  factory InventoryUsage.fromMap(Map<String, dynamic> map) {

class InventoryRepository {
  final AppDatabase _database = AppDatabase.instance;

  String _makeKey(String name, String unit) {
    return '${name.trim().toLowerCase()}|${unit.trim().toLowerCase()}';
  }

  double? _parseAmount(String value) {
    return double.tryParse(value.trim().replaceAll(',', '.'));
  }

  String _formatAmount(double value) {
    if (value == value.roundToDouble()) {
      return value.toInt().toString();
    }

    return value.toStringAsFixed(2).replaceFirst(RegExp(r'\.?0+$'), '');
  }

  Future<List<InventoryItem>> getAll() async {
    final db = await _database.database;

    final rows = await db.query(
      'inventory_items',
      orderBy: 'name COLLATE NOCASE ASC',
    );

    return rows.map(InventoryItem.fromMap).toList();
  }

  Future<List<InventoryUsage>> getRecentUsages({int limit = 10}) async {
    final db = await _database.database;

    final rows = await db.query(
      'inventory_usage',
      where: 'is_undone = 0',
      orderBy: 'used_at DESC',
      limit: limit,
    );

    return rows.map(InventoryUsage.fromMap).toList();
  }

  Future<void> add({
    required String name,
    required String amount,
    required String unit,
  }) async {
    final cleanName = name.trim();
    final cleanAmount = amount.trim();
    final cleanUnit = unit.trim();

    if (cleanName.isEmpty) return;

    final db = await _database.database;
    final key = _makeKey(cleanName, cleanUnit);

    await db.transaction((txn) async {
      final rows = await txn.query(
        'inventory_items',
        where: 'item_key = ?',
        whereArgs: [key],
        limit: 1,
      );

      if (rows.isEmpty) {
        await txn.insert('inventory_items', {
          'item_key': key,
          'name': cleanName,
          'amount': cleanAmount,
          'unit': cleanUnit,
        });
        return;
      }

      final oldAmount = rows.first['amount'] as String? ?? '';
      final oldNumber = _parseAmount(oldAmount);
      final newNumber = _parseAmount(cleanAmount);

      final combined = oldNumber != null && newNumber != null
          ? _formatAmount(oldNumber + newNumber)
          : cleanAmount.isNotEmpty
          ? cleanAmount
          : oldAmount;

      await txn.update(
        'inventory_items',
        {'amount': combined},
        where: 'item_key = ?',
        whereArgs: [key],
      );
    });
  }

  Future<void> consume(int id, double amountUsed) async {
    if (amountUsed <= 0) {
      throw ArgumentError('Amount used must be greater than zero.');
    }

    final db = await _database.database;

    await db.transaction((txn) async {
      final rows = await txn.query(
        'inventory_items',
        where: 'id = ?',
        whereArgs: [id],
        limit: 1,
      );

      if (rows.isEmpty) return;

      final item = rows.first;

      final currentAmount = _parseAmount(item['amount'] as String? ?? '');

      if (currentAmount == null) {
        throw StateError('This inventory item does not have a numeric amount.');
      }

      if (amountUsed > currentAmount) {
        throw ArgumentError('You cannot use more than the available amount.');
      }

      final itemKey = item['item_key'] as String;
      final name = item['name'] as String;
      final unit = item['unit'] as String? ?? '';

      // Save usage history before changing inventory.
      await txn.insert('inventory_usage', {
        'item_key': itemKey,
        'name': name,
        'amount': _formatAmount(amountUsed),
        'unit': unit,
        'used_at': DateTime.now().toIso8601String(),
        'is_undone': 0,
      });

      final remaining = currentAmount - amountUsed;

      if (remaining <= 0.000001) {
        await txn.delete('inventory_items', where: 'id = ?', whereArgs: [id]);
      } else {
        await txn.update(
          'inventory_items',
          {'amount': _formatAmount(remaining)},
          where: 'id = ?',
          whereArgs: [id],
        );
      }
    });
  }

  Future<void> undoUsage(int usageId) async {
    final db = await _database.database;

    await db.transaction((txn) async {
      final usageRows = await txn.query(
        'inventory_usage',
        where: 'id = ?',
        whereArgs: [usageId],
        limit: 1,
      );

      if (usageRows.isEmpty) return;

      final usage = usageRows.first;

      if ((usage['is_undone'] as int? ?? 0) == 1) return;

      final itemKey = usage['item_key'] as String;
      final name = usage['name'] as String;
      final unit = usage['unit'] as String? ?? '';
      final amount = usage['amount'] as String;

      final amountUsed = _parseAmount(amount);

      if (amountUsed == null || amountUsed <= 0) {
        throw StateError('Invalid usage amount.');
      }

      final inventoryRows = await txn.query(
        'inventory_items',
        where: 'item_key = ?',
        whereArgs: [itemKey],
        limit: 1,
      );

      if (inventoryRows.isEmpty) {
        await txn.insert('inventory_items', {
          'item_key': itemKey,
          'name': name,
          'amount': _formatAmount(amountUsed),
          'unit': unit,
        });
      } else {
        final oldAmount = inventoryRows.first['amount'] as String? ?? '';
        final oldNumber = _parseAmount(oldAmount);

        if (oldNumber == null) {
          throw StateError(
            'Cannot restore usage because the current inventory amount is not numeric.',
          );
        }

        await txn.update(
          'inventory_items',
          {'amount': _formatAmount(oldNumber + amountUsed)},
          where: 'item_key = ?',
          whereArgs: [itemKey],
        );
      }

      await txn.update(
        'inventory_usage',
        {'is_undone': 1},
        where: 'id = ?',
        whereArgs: [usageId],
      );
    });
  }

  Future<void> markUsed(int id) async {
    final db = await _database.database;

    await db.delete('inventory_items', where: 'id = ?', whereArgs: [id]);
  }

  Future<void> delete(int id) => markUsed(id);
}

class MealRepository {
  final AppDatabase _database = AppDatabase.instance;

  Future<List<Meal>> getAll() async {
    final db = await _database.database;

    final rows = await db.query('meals', orderBy: 'date ASC, id ASC');

    return rows
        .map((row) => Meal.fromMap(Map<String, dynamic>.from(row)))
        .toList();
  }

  Future<List<Meal>> getByDate(String date) async {
    final db = await _database.database;

    final rows = await db.query(
      'meals',
      where: 'date = ?',
      whereArgs: [date],
      orderBy: 'id ASC',
    );

    return rows
        .map((row) => Meal.fromMap(Map<String, dynamic>.from(row)))
        .toList();
  }

  Future<int> insert(Meal meal) async {
    final db = await _database.database;
    final values = meal.toMap()..remove('id');

    return db.insert('meals', values);
  }

  Future<void> update(Meal meal) async {
    if (meal.id == null) {
      throw ArgumentError('Meal ID cannot be null');
    }

    final db = await _database.database;
    final values = meal.toMap()..remove('id');

    await db.update('meals', values, where: 'id = ?', whereArgs: [meal.id]);
  }

  Future<void> delete(int id) async {

    await db.delete('meals', where: 'id = ?', whereArgs: [id]);

    List<RecipeIngredient> ingredients,
  ) async {
    var total = NutritionSummary.zero;

    for (final ingredient in ingredients) {
      if (ingredient.foodId == null) continue;
      if (ingredient.unit.toLowerCase() != 'g') continue;

      final grams = ingredient.grams;

      if (grams <= 0) continue;

      final food = await database.getFood(ingredient.foodId!);

      if (food == null) continue;

      total += NutritionSummary(
        calories: food.caloriesFor(grams),
        protein: food.proteinFor(grams),
        carbs: food.carbsFor(grams),
        fat: food.fatFor(grams),
      );
    }

    return total;
  }

  Future<NutritionSummary> forRecipe(Recipe recipe) async {
    return forIngredients(recipe.ingredients);
  }

  Future<NutritionSummary> forRecipeServing(Recipe recipe) async {
    final total = await forRecipe(recipe);
    final servings = recipe.servings > 0 ? recipe.servings : 1;

    return total / servings.toDouble();
  }

  Future<NutritionSummary> forMeal(Meal meal) async {
    if (meal.isRecipe) {
      if (meal.recipeId == null) return NutritionSummary.zero;

      final recipe = await recipeRepository.getById(meal.recipeId!);

      if (recipe == null) return NutritionSummary.zero;

      final servingNutrition = await forRecipeServing(recipe);
      final servings = meal.quantityValue;

      if (servings <= 0) return NutritionSummary.zero;

      return servingNutrition * servings;
    }

    if (meal.foodId == null || meal.unit.toLowerCase() != 'g') {
      return NutritionSummary.zero;
    }

    final grams = meal.quantityValue;

    if (grams <= 0) return NutritionSummary.zero;

    final SwissFood? food = await database.getFood(meal.foodId!);

    if (food == null) return NutritionSummary.zero;

    return NutritionSummary(
      calories: food.caloriesFor(grams),
      protein: food.proteinFor(grams),
      carbs: food.carbsFor(grams),
      fat: food.fatFor(grams),
    );
  }

  Future<NutritionSummary> forMeals(List<Meal> meals) async {
    var total = NutritionSummary.zero;

    for (final meal in meals) {
      total += await forMeal(meal);
    }

    return total;
  }
}

class NutritionSummary {
  final double calories;
  final double protein;
  final double carbs;
  final double fat;

  const NutritionSummary({
    required this.calories,
    required this.protein,
    required this.carbs,
    required this.fat,
  });

  static const zero = NutritionSummary(
    calories: 0,
    protein: 0,
    carbs: 0,
    fat: 0,
  );

  NutritionSummary operator +(NutritionSummary other) {
    return NutritionSummary(
      calories: calories + other.calories,
      protein: protein + other.protein,
      carbs: carbs + other.carbs,
      fat: fat + other.fat,
    );
  }

  NutritionSummary operator /(double divisor) {
    if (divisor == 0) return zero;

    return NutritionSummary(
      calories: calories / divisor,
      protein: protein / divisor,
      carbs: carbs / divisor,
      fat: fat / divisor,
    );
  }

  NutritionSummary operator *(double multiplier) {
    return NutritionSummary(
      calories: calories * multiplier,
      protein: protein * multiplier,
      carbs: carbs * multiplier,
      fat: fat * multiplier,
    );
  }

  String get formatted =>
      '${calories.round()} kcal · '
      'P ${protein.toStringAsFixed(1)} g · '
      'C ${carbs.toStringAsFixed(1)} g · '
      'F ${fat.toStringAsFixed(1)} g';
}

import 'package:sqflite/sqflite.dart';

import 'package:preplist/features/recipes/models/recipe.dart';
import 'package:preplist/core/database/app_database.dart';

class RecipeRepository {
  final AppDatabase _database = AppDatabase.instance;

  Future<List<Recipe>> getAll({String query = ''}) async {
    final db = await _database.database;
    final trimmed = query.trim();

    final rows = await db.query(
      'recipes',
      where: trimmed.isEmpty ? null : 'name LIKE ?',
      whereArgs: trimmed.isEmpty ? null : ['%$trimmed%'],
      orderBy: 'name COLLATE NOCASE ASC',
    );

    final recipes = <Recipe>[];

    for (final row in rows) {
      recipes.add(await _getRecipeWithIngredients(db, row));
    }

    return recipes;
  }

  Future<Recipe?> getById(int id) async {
    final db = await _database.database;

    final rows = await db.query(
      'recipes',
      where: 'id = ?',
      whereArgs: [id],
      limit: 1,
    );

    if (rows.isEmpty) return null;

    return _getRecipeWithIngredients(db, rows.first);
  }

  Future<Recipe> _getRecipeWithIngredients(
    DatabaseExecutor db,
    Map<String, Object?> row,
  ) async {
    final ingredientRows = await db.query(
      'recipe_ingredients',
      where: 'recipe_id = ?',
      whereArgs: [row['id']],
      orderBy: 'id ASC',
    );

    return Recipe.fromMap(
      Map<String, dynamic>.from(row),
      ingredients: ingredientRows
          .map(
            (ingredient) =>
                RecipeIngredient.fromMap(Map<String, dynamic>.from(ingredient)),
          )
          .toList(),
    );
  }

  Future<int> insert(Recipe recipe) async {
    final db = await _database.database;

    return db.transaction((txn) async {
      final recipeId = await txn.insert('recipes', recipe.toMap());

      for (final ingredient in recipe.ingredients) {
        await txn.insert('recipe_ingredients', {
          'recipe_id': recipeId,
          'food_id': ingredient.foodId,
          'name': ingredient.name,
          'amount': ingredient.amount,
          'unit': ingredient.unit,
        });
      }

      return recipeId;
    });
  }

  Future<void> update(Recipe recipe) async {
    final recipeId = recipe.id;

    if (recipeId == null) {
      throw ArgumentError('Recipe ID cannot be null');
    }

    final db = await _database.database;

    await db.transaction((txn) async {
      await txn.update(
        'recipes',
        recipe.toMap(),
        where: 'id = ?',
        whereArgs: [recipeId],
      );

      await txn.delete(
        'recipe_ingredients',
        where: 'recipe_id = ?',
        whereArgs: [recipeId],
      );

      for (final ingredient in recipe.ingredients) {
        await txn.insert('recipe_ingredients', {
          'recipe_id': recipeId,
          'food_id': ingredient.foodId,
          'name': ingredient.name,
          'amount': ingredient.amount,
          'unit': ingredient.unit,
        });
      }
    });
  }

  Future<void> delete(int id) async {
    final db = await _database.database;

    await db.delete('recipes', where: 'id = ?', whereArgs: [id]);
  }
}

class IngredientEditor extends StatefulWidget {
  final List<RecipeIngredient> ingredients;
  final ValueChanged<List<RecipeIngredient>> onChanged;

  const IngredientEditor({
    super.key,
    required this.ingredients,
    required this.onChanged,
  });

  @override
  State<IngredientEditor> createState() => _IngredientEditorState();
}

class _IngredientEditorState extends State<IngredientEditor> {
  late List<RecipeIngredient> _ingredients;

  @override
  void initState() {
    super.initState();
    _ingredients = List.of(widget.ingredients);
  }

  void _notifyChanged() {
    widget.onChanged(List.unmodifiable(_ingredients));
  }

  Future<void> _addIngredient() async {
    final food = await showDialog<SwissFood>(
      context: context,
      builder: (_) => const FoodPickerDialog(),
    );

    if (!mounted || food == null) return;

    final amount = await _askForAmount(title: food.name, initialAmount: '');

    if (!mounted || amount == null) return;

    final grams = _parseGrams(amount);

    if (grams == null || grams <= 0) {
      _showInvalidAmount();
      return;
    }

    setState(() {
      _ingredients.add(
        RecipeIngredient(
          foodId: food.id,
          name: food.name,
          amount: amount,
          unit: 'g',
        ),
      );
    });

    _notifyChanged();
  }

  Future<void> _editIngredient(int index) async {
    if (index < 0 || index >= _ingredients.length) return;

    final ingredient = _ingredients[index];

    final amount = await _askForAmount(
      title: ingredient.name,
      initialAmount: ingredient.amount,
    );

    if (!mounted || amount == null) return;

    final grams = _parseGrams(amount);

    if (grams == null || grams <= 0) {
      _showInvalidAmount();
      return;
    }

    if (index >= _ingredients.length ||
        _ingredients[index].id != ingredient.id) {
      return;
    }

    setState(() {
      _ingredients[index] = RecipeIngredient(
        id: ingredient.id,
        recipeId: ingredient.recipeId,
        foodId: ingredient.foodId,
        name: ingredient.name,
        amount: amount,
        unit: ingredient.unit,
      );
    });

    _notifyChanged();
  }

  Future<String?> _askForAmount({
    required String title,
    required String initialAmount,
  }) {
    return showDialog<String>(
      context: context,
      builder: (_) =>
          _IngredientAmountDialog(title: title, initialAmount: initialAmount),
    );
  }

  double? _parseGrams(String value) {
    return double.tryParse(value.replaceAll(',', '.'));
  }

  void _showInvalidAmount() {
    ScaffoldMessenger.of(context).showSnackBar(
      const SnackBar(
        content: Text('Enter a valid amount greater than zero grams.'),
      ),
    );
  }

  void _removeIngredient(int index) {
    setState(() {
      _ingredients.removeAt(index);
    });

    _notifyChanged();
  }

  @override
  Widget build(BuildContext context) {
    final theme = Theme.of(context);

    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        SizedBox(
          width: double.infinity,
          child: OutlinedButton.icon(
            onPressed: _addIngredient,
            icon: const Icon(Icons.add),
            label: const Text('Add food from Swiss database'),
          ),
        ),
        const SizedBox(height: 16),
        Row(
          children: [
            Text(
              'Ingredients',
              style: theme.textTheme.titleSmall?.copyWith(
                fontWeight: FontWeight.bold,
              ),
            ),
            const SizedBox(width: 8),
            Text(
              '${_ingredients.length}',
              style: theme.textTheme.labelMedium?.copyWith(
                color: theme.colorScheme.onSurfaceVariant,
              ),
            ),
          ],
        ),
        const SizedBox(height: 8),
        if (_ingredients.isEmpty)
          Padding(
            padding: const EdgeInsets.symmetric(vertical: 16),
            child: Center(
              child: Text(
                'No ingredients added yet',
                style: theme.textTheme.bodyMedium?.copyWith(
                  color: theme.colorScheme.onSurfaceVariant,
                ),
              ),
            ),
          )
        else
          ...List.generate(_ingredients.length, (index) {
            final ingredient = _ingredients[index];

            return Card(
              margin: const EdgeInsets.only(bottom: 6),
              child: ListTile(
                dense: true,
                title: Text(ingredient.name),
                subtitle: Text('${ingredient.amount} g'),
                leading: const Icon(Icons.eco_outlined),
                trailing: Row(
                  mainAxisSize: MainAxisSize.min,
                  children: [
                    IconButton(
                      tooltip: 'Edit ingredient',
                      icon: const Icon(Icons.edit_outlined),
                      onPressed: () => _editIngredient(index),
                    ),
                    IconButton(
                      tooltip: 'Remove ingredient',

                      onPressed: () => _removeIngredient(index),
                    ),
                  ],
                ),
              ),
            );
          }),
      ],
    );
  }

class PrepListApp extends flutter.StatelessWidget {
  const PrepListApp({super.key});

  static const _seedColor = flutter.Colors.green;

  flutter.ColorScheme _fallbackScheme(flutter.Brightness brightness) {
    return flutter.ColorScheme.fromSeed(
      seedColor: _seedColor,
      brightness: brightness,
    );
  }

  flutter.ColorScheme _convertScheme(
    dynamic source,
    flutter.Brightness brightness,
  ) {
    if (source == null) {
      return _fallbackScheme(brightness);
    }

    return flutter.ColorScheme(
      brightness: brightness,
      primary: source.primary,
      onPrimary: source.onPrimary,
      primaryContainer: source.primaryContainer,
      onPrimaryContainer: source.onPrimaryContainer,
      secondary: source.secondary,
      onSecondary: source.onSecondary,
      secondaryContainer: source.secondaryContainer,
      onSecondaryContainer: source.onSecondaryContainer,
      tertiary: source.tertiary,
      onTertiary: source.onTertiary,
      tertiaryContainer: source.tertiaryContainer,
      onTertiaryContainer: source.onTertiaryContainer,
      error: source.error,
      onError: source.onError,
      errorContainer: source.errorContainer,
      onErrorContainer: source.onErrorContainer,
      surface: source.surface,
      onSurface: source.onSurface,
      surfaceContainerLowest: source.surfaceContainerLowest,
      surfaceContainerLow: source.surfaceContainerLow,
      surfaceContainer: source.surfaceContainer,
      surfaceContainerHigh: source.surfaceContainerHigh,
      surfaceContainerHighest: source.surfaceContainerHighest,
      onSurfaceVariant: source.onSurfaceVariant,
      outline: source.outline,
      outlineVariant: source.outlineVariant,
      shadow: source.shadow,
      scrim: source.scrim,
      inverseSurface: source.inverseSurface,
      onInverseSurface: source.onInverseSurface,
      inversePrimary: source.inversePrimary,
    );
  }

  @override
  flutter.Widget build(flutter.BuildContext context) {
    return DynamicColorBuilder(
      builder: (lightDynamic, darkDynamic) {
        final lightScheme = _convertScheme(
          lightDynamic,
          flutter.Brightness.light,
        );

        final darkScheme = _convertScheme(darkDynamic, flutter.Brightness.dark);

        return flutter.MaterialApp(
          title: 'PrepList',
          debugShowCheckedModeBanner: false,
          theme: flutter.ThemeData(
            useMaterial3: true,
            colorScheme: lightScheme,
          ),
          darkTheme: flutter.ThemeData(
            useMaterial3: true,
            colorScheme: darkScheme,
          ),
          themeMode: flutter.ThemeMode.system,
          home: const FoodBootstrapper(child: MainScreen()),
        );
      },
    );
  }
}
